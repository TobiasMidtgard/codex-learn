## Where JavaScript goes

```html
<script src="app.js"><\/script>   <\!-- just before </body>, so the page exists when it runs -->
```

Open the browser console (F12 → Console) — `console.log(...)` prints there, and errors show up with line numbers.

## Variables

```js
const taxRate = 0.25;      // cannot be reassigned — use this by default
let total = 0;             // can be reassigned
total = total + 10;
```

Forget `var`; it exists for old code.

## Types and strings

```js
typeof 42          // "number"   (integers and decimals are the same type)
typeof "hi"        // "string"
typeof true        // "boolean"
typeof undefined   // "undefined" — a variable with no value yet
null               // "no value", on purpose

const name = "Ada";
console.log(`Hello, ${name}! Total: ${total * 2}`);   // template literal: backticks + ${}
```

Always compare with `===` (and `!==`). The two-equals version converts types and surprises people: `"5" == 5` is `true`.

## Functions

```js
function greet(name) {
  return `Hello, ${name}`;
}

const double = (n) => n * 2;                 // arrow function: implicit return for one expression
const describe = (item) => {                 // braces when you need statements
  const label = item.name.toUpperCase();
  return `${label}: ${item.price}`;

```

## Arrays

```js
const prices = [120, 899, 249];
prices.push(15);
prices.length;                          // 4
prices.includes(899);                   // true

prices.map(p => p * 1.25);              // new array, each transformed
prices.filter(p => p > 200);            // new array, only those that pass
prices.reduce((sum, p) => sum + p, 0);  // fold into one value: 1283
prices.find(p => p > 200);              // first match or undefined
prices.forEach(p => console.log(p));    // loop with no result
```

`map`, `filter` and `reduce` return new arrays and never change the original. Chain them.

## Objects

```js
const item = { name: "Jack", price: 899, qty: 2 };
item.price;                 // 899
item["name"];               // "Jack"
item.qty = 3;               // update
const { name, price } = item;   // destructuring into variables

for (const [key, value] of Object.entries(item)) { console.log(key, value); }
```

## Common mistakes

- `Uncaught ReferenceError: x is not defined` — typo or the variable isn't in scope.
- `Cannot read properties of undefined` — you called `.something` on a value that doesn't exist yet (often a `querySelector` that found nothing).
- Forgetting `return` in a function with braces.

function cartTotal(items) {
  return items.reduce((sum, item) => sum + item.price * item.qty, 0);
}

function formatPrice(amount) {
  return `$${amount.toFixed(2)}`;
}

function applyDiscount(total, code) {
  const normalized = (code || "").toUpperCase();
  if (normalized === "SAVE10") return total * 0.9;
  if (normalized === "HALF") return total * 0.5;
  return total;
}

function inStock(items) {

## Finding elements

The DOM is the browser's live model of your HTML. JavaScript reads and changes it.

```js
const title = document.querySelector("h1");          // first match, CSS selector syntax
const button = document.querySelector("#add");       // by id
const items = document.querySelectorAll(".item");    // all matches (a NodeList — loop with forEach or for...of)
```

`querySelector` returns `null` when nothing matches. Guard against it while you're getting a page set up.

## Reading and changing content

```js
title.textContent = "Inventory";          // text only — safe for anything a user typed
title.innerHTML = "<em>Inventory</em>";   // parses HTML — never put user input here
input.value                               // what's typed in an <input>
img.src = "photo.jpg";                    // attributes are properties
el.classList.add("done");                 // add / remove / toggle / contains
el.style.color = "tomato";                // inline style (prefer classes)
el.dataset.id                             // data-id="…" attributes
```

## Creating and removing elements

```js
const li = document.createElement("li");
li.textContent = "Torch";
list.append(li);          // add to the end (prepend for the start)
li.remove();              // delete it
list.innerHTML = "";      // empty a container
```

## Events

```js
button.addEventListener("click", () => {
  count += 1;
  render();
});

input.addEventListener("input", (event) => {
  console.log(event.target.value);   // the element that fired the event
});

form.addEventListener("submit", (event) => {
  event.preventDefault();            // stop the page from reloading
  addTask(input.value);
});
```

Listen for `click` on buttons, `input` on text fields as the user types, `change` on selects and checkboxes, `submit` on forms, `keydown` for keyboard shortcuts.

## The state → render pattern

Keep your data in plain JavaScript variables and write one function that draws the page from that data. Events change the data and call `render()`. This scales from a counter to a whole app.

```js
let tasks = [];

function render() {
  list.innerHTML = "";
  for (const task of tasks) {
    const li = document.createElement("li");
    li.textContent = task.title;
    li.classList.toggle("done", task.done);
    list.append(li);
  }
  remaining.textContent = tasks.filter(t => !t.done).length;
}
```

## Event delegation

Instead of a listener per item, put one on the container and look at `event.target`:

```js
list.addEventListener("click", (event) => {
  const li = event.target.closest("li");
  if (!li) return;
  toggle(li.dataset.id);
});
```

This keeps working for elements you add later.

## Some things take time

Reading a file, asking a server, waiting for a timer — the browser refuses to freeze the page while it waits. Instead the work runs *asynchronously*: you hand over a function to call **later**.

```js
setTimeout(() => console.log("two seconds later"), 2000);
console.log("this prints first");
```

## Promises

A **Promise** is an object standing in for a value that isn't ready yet. It will either *resolve* with the value or *reject* with an error.

```js
fetch("/api/users")
  .then(response => response.json())
  .then(users => console.log(users))
  .catch(error => console.error("Failed:", error));
```

Each `.then` runs when the previous step finishes and passes its result along.

## async / await: the readable version

```js
async function loadUsers() {
  try {
    const response = await fetch("/api/users");
    if (!response.ok) throw new Error(`HTTP ${response.status}`);
    const users = await response.json();
    return users;
  } catch (error) {
    console.error("Failed:", error);
    return [];
  }
}
```

- `await` pauses *this function* until the promise settles, without freezing the page.
- `await` only works inside a function marked `async`.
- An `async` function always returns a promise — so callers `await loadUsers()` too.

## fetch and JSON

`fetch(url)` sends an HTTP request and resolves with a `Response`. `response.json()` parses the body — it is *also* asynchronous. A 404 does **not** reject the promise; check `response.ok` yourself.

Sending data:

```js
await fetch("/api/todos", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ title: "Buy oil" }),
});
```

## The UI pattern

1. Show a loading state.
2. Await the data.
3. Render it — or render the error.

```js
status.textContent = "Loading…";
const users = await loadUsers();
render(users);
status.textContent = `Loaded ${users.length} users`;
```

## CORS, in one paragraph

Browsers only let a page call servers that explicitly allow it (via the `Access-Control-Allow-Origin` header). When a `fetch` to someone else's API fails with a CORS error, it isn't your code — the server hasn't opted in. You'll learn to set that header in the backend track.

    for (const card of inColumn) {
      const li = document.createElement("li");
      li.className = "card";
      li.dataset.id = card.id;

      const title = document.createElement("span");
      title.className = "title";
      title.textContent = card.title;
      li.append(title);

      if (status !== "done") {
        const move = document.createElement("button");
        move.className = "move";
        move.textContent = "→";
        move.title = "Move to next column";
        li.append(move);
      }

      const del = document.createElement("button");
      del.className = "delete";
      del.textContent = "✕";
      del.title = "Delete";
      li.append(del);

      list.append(li);
    }
    column.querySelector(".count").textContent = inColumn.length;
  }
}

form.addEventListener("submit", (event) => {
  event.preventDefault();
  const title = input.value.trim();
  if (!title) return;
  cards.push({ id: nextId++, title, status: "todo" });
  input.value = "";
  render();
});

board.addEventListener("click", (event) => {
  const cardEl = event.target.closest(".card");
  if (!cardEl) return;
  const card = cards.find(c => c.id === Number(cardEl.dataset.id));
  if (!card) return;

  if (event.target.closest(".move")) {
    const next = ORDER[ORDER.indexOf(card.status) + 1];
    if (next) card.status = next;
  } else if (event.target.closest(".delete")) {
    cards = cards.filter(c => c !== card);
  }
  render();
});

render();
